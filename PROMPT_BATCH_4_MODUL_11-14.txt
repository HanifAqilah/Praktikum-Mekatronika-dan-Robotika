   from rclpy.node import Node

   class MqttBridge(Node):
       def __init__(self):
           super().__init__('mqtt_bridge')
           self.publisher = self.create_publisher(Imu, '/robot/sensor/imu', 10)
           
           # Setup MQTT
           self.mqtt_client = mqtt.Client()
           self.mqtt_client.on_connect = self.on_mqtt_connect
           self.mqtt_client.on_message = self.on_mqtt_message
           self.mqtt_client.connect("localhost", 1883, 60)
           self.mqtt_client.loop_start()
       
       def on_mqtt_connect(self, client, userdata, flags, rc):
           self.get_logger().info(f"MQTT connected with result code {rc}")
           client.subscribe("robot/sensor/imu")
       
       def on_mqtt_message(self, client, userdata, msg):
           # Parse JSON
           payload = json.loads(msg.payload.decode())
           
           # Convert to ROS2 message
           imu_msg = Imu()
           imu_msg.header.stamp = self.get_clock().now().to_msg()
           imu_msg.linear_acceleration.x = payload['data']['accel_x']
           
           # Publish
           self.publisher.publish(imu_msg)
           self.get_logger().debug(f"Published IMU: {payload}")
   ```

   def send_command_with_retry(self, command_dict, max_retry=3):
       for attempt in range(max_retry):
           # Send command
           seq = self.seq_counter
           self.seq_counter += 1
           command_dict['seq'] = seq
           message = json.dumps(command_dict)
           self.mqtt_client.publish("robot/command", message)
           
           # Wait for ACK (timeout 1 sec)
           start_time = time.time()
           while time.time() - start_time < 1.0:
               if seq in self.ack_received:
                   self.get_logger().info(f"Command {seq} acknowledged")
                   return True
               time.sleep(0.01)
           
           self.get_logger().warn(f"Command {seq} timeout, retry {attempt+1}")
       
       return False  # All retry failed
   ```

   Python code:
   ```python
   from abc import ABC, abstractmethod
   from enum import Enum
   from queue import Queue
   import time
   
   class StateType(Enum):
       IDLE = 1
       HOMING = 2
       READY = 3
       MOVING = 4
       WORKING = 5
       ERROR = 6
   
   class State(ABC):
       def __init__(self, state_type):
           self.type = state_type
       
       def on_enter(self):
           """Execute when entering state"""
           pass
       
       def on_exit(self):
           """Execute when leaving state"""
           pass
       
       def handle_event(self, event):
           """Handle event in this state"""
           return None  # return next state or None if no transition
   
   class IdleState(State):
       def __init__(self):
           super().__init__(StateType.IDLE)
       
       def on_enter(self):
           print("Entering IDLE state")
           # Disable motors, stop publishing
       
       def handle_event(self, event):
           if event['type'] == 'start_command':
               if event.get('motor_enabled', False):
                   return StateType.HOMING
           return None
   
   class HomingState(State):
       def __init__(self):
           super().__init__(StateType.HOMING)
       
       def on_enter(self):
           print("Entering HOMING state")
           # Execute homing sequence
       
       def handle_event(self, event):
           if event['type'] == 'reached_home':
               return StateType.READY
           elif event['type'] == 'fault_detected':
               return StateType.ERROR
           return None
   
   class StateMachine:
       def __init__(self):
           self.states = {
               StateType.IDLE: IdleState(),
               StateType.HOMING: HomingState(),
               # ... other state ...
           }
           self.current_state = self.states[StateType.IDLE]
           self.event_queue = Queue()
           self.running = True
       
       def post_event(self, event):
           """Add event to queue"""
           self.event_queue.put(event)
       
       def process_events(self):
           """Process one event from queue"""
           if not self.event_queue.empty():
               event = self.event_queue.get()
               next_state_type = self.current_state.handle_event(event)
               
               if next_state_type:
                   self.transition_to(next_state_type)
       
       def transition_to(self, next_state_type):
           """Execute state transition"""
           self.current_state.on_exit()
           self.current_state = self.states[next_state_type]
           self.current_state.on_enter()
           print(f"Transitioned to {next_state_type}")
       
       def run(self):
           """Main loop"""
           while self.running:
               self.process_events()
               time.sleep(0.01)  # 100 Hz
   ```

       def __init__(self, event_type, condition, next_state, action=None):
           self.event_type = event_type
           self.condition = condition  # Lambda function, return True/False
           self.next_state = next_state
           self.action = action  # Function to execute during transition
   
   class EnhancedState(State):
       def __init__(self, state_type):
           super().__init__(state_type)
           self.transitions = []
       
       def add_transition(self, event_type, condition, next_state, action=None):
           self.transitions.append(GuardedTransition(event_type, condition, next_state, action))
       
       def handle_event(self, event, context):
           for trans in self.transitions:
               if event['type'] == trans.event_type:
                   if trans.condition(context):  # Check guard
                       if trans.action:
                           trans.action(context)  # Execute action
                       return trans.next_state
           return None

   PERCOBAAN 4: KPI DEFINITION & CALCULATION
   - Define 3 KPI: robot availability, cycle time average, energy efficiency
   - Calculate KPI per day, per week
   - Store KPI di InfluxDB
   
   KPI 1: Robot availability
   ```
   Availability = sum(uptime) / total_time × 100%
   
   Calculation:
   - Get all "robot_state" metric
   - Filter state='running' → sum duration
   - Total duration = 24 hours
   - Availability = running_time / 24h
   ```
   
   KPI 2: Cycle time average
   ```
   CycleTime = (task_end_time - task_start_time)
   AvgCycleTime = sum(cycle_time) / number_of_cycle
   ```
   
   KPI 3: Energy efficiency
   ```
   Energy_per_cycle = total_energy / number_of_cycle
   ```
   
   Python script to calculate & store KPI:
   ```python
   from influxdb_client import InfluxDBClient
   from datetime import datetime, timedelta
   
   def calculate_kpi_availability():
       # Query running time
       query = '''
       from(bucket: "my-bucket")
         |> range(start: -1d)
         |> filter(fn: (r) => r["_measurement"] == "robot_state")
         |> filter(fn: (r) => r["state"] == "running")
         |> aggregateWindow(every: 1d, fn: sum)
       '''
       result = client.query_api().query(org="my-org", query=query)
       
       running_time = 0
       for table in result:
           for record in table.records:
               running_time = record["_value"]
       
       availability = (running_time / (24*3600)) * 100  # Assume 24h = total
       
       # Store KPI
       point = influxdb_client.Point("kpi_availability") \
           .field("percentage", availability) \
           .time(datetime.utcnow())
       
       write_api.write(bucket="my-bucket", record=point)
       
       return availability
   ```
